-----------------------------CREATE TABLE
CREATE TABLE PaymentsTable (
    PaymentId nvarchar(50) NOT NULL PRIMARY KEY,
    MemberId nvarchar(50) NOT NULL,
    PlanId nvarchar(50) NOT NULL,
    Amount int NOT NULL,
    PaymentDate datetime NOT NULL,
    PaymentStatus nvarchar(50) NOT NULL, 
    PaymentMethod nvarchar(50) NOT NULL, 
    TransactionId nvarchar(100) NULL,
    CONSTRAINT FK_PaymentsTable_MemberId_MembersTable_MemberId FOREIGN KEY (MemberId) REFERENCES MembersTable(MemberId),
    CONSTRAINT FK_PaymentsTable_PlanId_PlansTable_PlanId FOREIGN KEY (PlanId) REFERENCES PlansTable(PlanId)
);
GO

-----------------------------CREATE MODULE
public class PaymentTable
    {
        [Key]
        public string PaymentId { get; set; }

        public string MemberId { get; set; }

        public string PlanId { get; set; }

    
        public int Amount { get; set; }

      
        public DateTime PaymentDate { get; set; }

        public string PaymentStatus { get; set; }

 
        public string PaymentMethod { get; set; }

        public string TransactionId { get; set; }

        [ForeignKey("MemberId")]
        public virtual MemberTable Member { get; set; }

        [ForeignKey("PlanId")]
        public virtual PlanTable Plan { get; set; }
    }

------------------------CONTROLLER
using GymManagementSystem.Models;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Web.Mvc;

namespace GymManagementSystem.Controllers
{
    public class PaymentController : Controller
    {
        public ApplicationContextDBConnect db = new ApplicationContextDBConnect();

        public ActionResult BuyPlan(string planId)
        {
            var plan = db.PlansTable.FirstOrDefault(p => p.PlanId == planId && p.IsActive);
            if (plan == null)
                return HttpNotFound();

            return View(plan);
        }

        [HttpPost]
        public ActionResult ProcessPayment(string planId, string paymentMethod)
        {
            var userId = Session["UserId"].ToString(); //session logic
            var member = db.MembersTable.FirstOrDefault(m => m.UserId == userId);

            if (member == null) 
                return HttpNotFound("Member not found.");

            var plan = db.PlansTable.FirstOrDefault(p => p.PlanId == planId);
            if (plan == null) 
                return HttpNotFound("Plan not found.");

            // Simulate payment
            var paymentSuccess = true; 

            if (paymentSuccess)
            {
                var payment = new Payment
                {
                    PaymentId = Guid.NewGuid().ToString(),
                    MemberId = member.MemberId,
                    PlanId = plan.PlanId,
                    Amount = plan.PlanPrice,
                    PaymentDate = DateTime.Now,
                    PaymentStatus = "Success",
                    PaymentMethod = paymentMethod,
                    TransactionId = Guid.NewGuid().ToString()
                };

                member.PlanId = plan.PlanId;
                member.PlanStart = DateTime.Now;
                member.PlanEnd = DateTime.Now.AddDays(plan.PlanDuration);

                db.PaymentTable.Add(payment);
                db.SaveChanges();
                ViewBag.Message = "Payment Successful";
                return RedirectToAction("LoginIndex", "UserLogin");
            }
            else
            {
                return RedirectToAction("BuyPlan", "Failed Payment");
            }
        }
    }

}
