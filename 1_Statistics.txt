    {
        public int TotalTrainers { get; set; }
        public int VerifiedTrainers { get; set; }
        public int UnverifiedTrainers { get; set; }
        public Dictionary<string, int> GenderDistribution { get; set; }
        public double AverageExperience { get; set; }
        public Dictionary<string, int> SpecialityCounts { get; set; }
        public Dictionary<string, int> AgeGroups { get; set; }
    }


-- Add Controller - Action Method:
public ActionResult TrainerStats()
        {
            using (var db = new ApplicationContextDBConnect())
            {
                var stats = new TrainerStatisticsViewModel
                {
                    TotalTrainers = db.TrainerInformationTable.Count(),
                    VerifiedTrainers = db.TrainerInformationTable.Count(t => t.TrainerIsVerified),
                    UnverifiedTrainers = db.TrainerInformationTable.Count(t => !t.TrainerIsVerified),
                    GenderDistribution = db.TrainerInformationTable
                                          .GroupBy(t => t.TrainerGender)
                                          .ToDictionary(g => g.Key, g => g.Count()),
                    AverageExperience = db.TrainerInformationTable.Average(t => t.TrainerExperience),
                    SpecialityCounts = db.TrainerInformationTable
                                         .GroupBy(t => t.TrainerSpecialities)
                                         .ToDictionary(g => g.Key, g => g.Count()),
                    AgeGroups = db.TrainerInformationTable.AsEnumerable().GroupBy(t =>
                                  {
                                      int age = DateTime.Now.Year - t.TrainerDOB.Year;
                                      if (t.TrainerDOB.Date > DateTime.Now.AddYears(-age)) age--;

                                      if (age < 25) return "Under 25";
                                      if (age < 35) return "25-34";
                                      if (age < 45) return "35-44";
                                      return "45+";
                                  })
                                  .ToDictionary(g => g.Key, g => g.Count())
                };

                return View(stats);
            }
        }

public class MemberStatisticsViewModel
{
    public int TotalMembers { get; set; }
    public int VerifiedMembers { get; set; }
    public int UnverifiedMembers { get; set; }

    public int ActivePlans { get; set; }
    public int InactivePlans { get; set; }

    public int MembersWithTrainer { get; set; }
    public int MembersWithDietician { get; set; }

    public double AverageHeight { get; set; }
    public double AverageWeight { get; set; }

    public Dictionary<string, int> BloodGroupDistribution { get; set; }
    public Dictionary<string, int> PlansStartedPerMonth { get; set; }
    public double AveragePlanDurationDays { get; set; }
}


-- Create Controller: 
public ActionResult MemberStats()
{
    using (var db = new YourDbContext())
    {
        var members = db.MembersTable.ToList();

        var stats = new MemberStatisticsViewModel
        {
            TotalMembers = members.Count,
            VerifiedMembers = members.Count(m => m.MemberIsVerified),
            UnverifiedMembers = members.Count(m => !m.MemberIsVerified),

            ActivePlans = members.Count(m => m.PlanStatus),
            InactivePlans = members.Count(m => !m.PlanStatus),

            MembersWithTrainer = members.Count(m => !string.IsNullOrEmpty(m.TrainerId)),
            MembersWithDietician = members.Count(m => !string.IsNullOrEmpty(m.DieticianId)),

            AverageHeight = members.Any() ? members.Average(m => m.MemberHeight) : 0,
            AverageWeight = members.Any() ? members.Average(m => m.MemberWeight) : 0,

            BloodGroupDistribution = members
                .GroupBy(m => m.MemberBloodGroup)
                .ToDictionary(g => g.Key, g => g.Count()),

            PlansStartedPerMonth = members
                .Where(m => m.PlanStart != null)
                .GroupBy(m => m.PlanStart.Value.ToString("yyyy-MM"))
                .OrderBy(g => g.Key)
                .ToDictionary(g => g.Key, g => g.Count()),

            AveragePlanDurationDays = members
                .Where(m => m.PlanStart != null && m.PlanEnd != null)
                .Select(m => (m.PlanEnd.Value - m.PlanStart.Value).TotalDays)
                .DefaultIfEmpty(0)
                .Average()
        };

        return View(stats);
    }
}
