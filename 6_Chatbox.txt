CREATE TABLE Chat (
MessageId NVARCHAR(50) NOT NULL,
MessageTo NVARCHAR(50) NOT NULL,
MessageFrom	NVARCHAR(50) NOT NULL,
ChatMessage NVARCHAR(1000) NOT NULL,
MessagePostedOn DATETIME NOT NULL DEFAULT GETDATE(),
CONSTRAINT PK_Chat_MessageId PRIMARY KEY (MessageId)
);

    <script>
        const chatMessages = document.getElementById("chatMessages");
        const messageInput = document.getElementById("messageInput");

        function sendMessage() {
            const message = messageInput.value.trim();
            if (message === "") return;

            // Append sent message
            const messageElem = document.createElement("div");
            messageElem.classList.add("message", "sent");
            messageElem.innerHTML = `<p>${message}</p>`;
            chatMessages.appendChild(messageElem);

            // Clear input
            messageInput.value = "";

            // Scroll to bottom
            chatMessages.scrollTop = chatMessages.scrollHeight;

            // Simulate received message (for demo)
            setTimeout(() => {
                const reply = document.createElement("div");
                reply.classList.add("message", "received");
                reply.innerHTML = `<p>Echo: ${message}</p>`;
                chatMessages.appendChild(reply);
                chatMessages.scrollTop = chatMessages.scrollHeight;
            }, 1000);
        }

        // Enter key support
        messageInput.addEventListener("keydown", function (e) {
            if (e.key === "Enter") {
                sendMessage();
            }
        });
    </script>
